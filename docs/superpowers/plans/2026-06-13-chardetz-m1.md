- [ ] **Step 3: Implement src/state_machine.zig**

Port uchardet's `nsPkgInt.h` packing. `GETFROMPCK(i, x)` in uchardet is:
`((((x).data[(i)>>(x).idxsft]) >> (((i)&(x).sftmsk)<<(x).bitsft)) & (x).unitmsk)`.

```zig
// (prepend LICENSE_HEADER)
const std = @import("std");

pub const eIdxSft4bits: u32 = 3;
pub const eIdxSft8bits: u32 = 2;
pub const eIdxSft16bits: u32 = 1;
pub const eSftMsk4bits: u32 = 7;
pub const eSftMsk8bits: u32 = 3;
pub const eSftMsk16bits: u32 = 1;
pub const eBitSft4bits: u32 = 2;
pub const eBitSft8bits: u32 = 3;
pub const eBitSft16bits: u32 = 4;
pub const eUnitMsk4bits: u32 = 0x0000000F;
pub const eUnitMsk8bits: u32 = 0x000000FF;
pub const eUnitMsk16bits: u32 = 0x0000FFFF;

/// Bit-packed integer table (uchardet nsPkgInt). Packs 4/8/16-bit units into
/// u32 words; `get(i)` is the Zig translation of uchardet's GETFROMPCK macro.
pub const PckInt = struct {
    idx_sft: u32,
    sft_msk: u32,
    bit_sft: u32,
    unit_msk: u32,
    data: []const u32,
    pub fn get(self: PckInt, i: u32) u32 {
        const word = self.data[i >> @intCast(self.idx_sft)];
        const shift: u5 = @intCast((i & self.sft_msk) << @intCast(self.bit_sft));
        return (word >> shift) & self.unit_msk;
    }
};

/// uchardet SMModel: class table (byte→class), class count, state table, and
/// per-class char length. Drives a coding state machine for multibyte probers.
pub const SMModel = struct {
    class_table: PckInt,
    class_factor: u32,
    state_table: PckInt,
    char_len_table: []const u32,
    name: []const u8,
};
```
